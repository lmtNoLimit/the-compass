export async function action(args: Route.ActionArgs) {
  try {
    const { userId } = await getAuth(args.request); // Clerk integration
    const body = await args.request.json();
    const { action, agentId = 'demo-agent' } = body;
    
    switch (action) {
      case 'create_session':
        return vertexAI.createAgentSession(userId, agentId);
      case 'query_session':
        return vertexAI.streamQuery({ ...body, userId, agentId });
      case 'list_conversations':
        return conversationService.getByUser(userId, agentId);
      case 'delete_conversation':
        return conversationService.delete(body.sessionId, userId);
    }
  } catch (error) {
    return json({ error: error.message }, { status: 500 });
  }
}

import pytest
from app.agent import EnterpriseAdminAgent

class TestEnterpriseAdminAgent:
    @pytest.fixture
    def agent(self):
        return EnterpriseAdminAgent()
    
    def test_persona_consistency(self, agent):
        """Test that agent maintains enterprise admin persona."""
        prompts = [
            "What are your main concerns about new software?",
            "How do you evaluate security risks?",
            "What's your budget approval process?"
        ]
        
        for prompt in prompts:
            response = agent.process_prompt(prompt)
            # Check for enterprise admin characteristics
            assert any(keyword in response.lower() for keyword in [
                'security', 'compliance', 'risk', 'enterprise', 'policy'
            ])
    
    def test_temperature_consistency(self, agent):
        """Test response consistency with proper temperature setting."""
        prompt = "Tell me about your role."
        responses = [agent.process_prompt(prompt) for _ in range(3)]
        
        # Responses should be similar but not identical
        assert len(set(responses)) > 1  # Some variation
        # But all should contain enterprise admin context
        for response in responses:
            assert 'administrator' in response.lower() or 'admin' in response.lower()
